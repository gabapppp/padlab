cmake_minimum_required(VERSION 3.16)
project(PadLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets Concurrent)
set(QT Qt5)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(padcore src/pipeline.cpp src/advanced.cpp src/dataset.cpp)
target_include_directories(padcore PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(padcore PUBLIC ${QT}::Core ${OpenCV_LIBS})
add_executable(padlab src/main.cpp src/window.cpp)
target_link_libraries(padlab PRIVATE padcore ${QT}::Widgets ${QT}::Concurrent)
add_executable(padlab-cli src/cli.cpp)
target_link_libraries(padlab-cli PRIVATE padcore)
include(CTest)
if(BUILD_TESTING)
  add_executable(padcore-tests tests/core_tests.cpp)
  target_link_libraries(padcore-tests PRIVATE padcore)
  add_test(NAME padcore-tests COMMAND padcore-tests)
  add_executable(padadvanced-tests tests/advanced_tests.cpp)
  target_link_libraries(padadvanced-tests PRIVATE padcore)
  add_test(NAME padadvanced-tests COMMAND padadvanced-tests)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  add_executable(padgui-tests tests/gui_tests.cpp src/window.cpp)
  target_link_libraries(padgui-tests PRIVATE padcore Qt5::Widgets Qt5::Concurrent Qt5::Test)
  add_test(NAME padgui-tests COMMAND padgui-tests)
  set_tests_properties(padgui-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(padlab-demo tests/demo.cpp)
  target_link_libraries(padlab-demo PRIVATE padcore)
  add_test(NAME generate-demo COMMAND padlab-demo ${CMAKE_BINARY_DIR}/demo)
  set_tests_properties(generate-demo PROPERTIES FIXTURES_SETUP demo)
  add_test(NAME cli-demo COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:padlab-cli> -DDEMO=${CMAKE_BINARY_DIR}/demo
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.cmake)
  set_tests_properties(cli-demo PROPERTIES FIXTURES_REQUIRED demo)
endif()
install(TARGETS padlab padlab-cli RUNTIME DESTINATION bin)

install(FILES packaging/padlab.desktop DESTINATION share/applications)
install(FILES packaging/padlab.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES packaging/INSTALL.txt DESTINATION share/doc/padlab)
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "padlab")
set(CPACK_PACKAGE_VENDOR "PadLab")
set(CPACK_PACKAGE_CONTACT "PadLab maintainers")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "OpenCV pipeline workbench with subpixel boundary measurements")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "PadLab maintainers")
set(CPACK_DEBIAN_PACKAGE_SECTION "graphics")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qtwayland5")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_FILE_NAME "padlab-${PROJECT_VERSION}-ubuntu22.04-amd64")
include(CPack)
